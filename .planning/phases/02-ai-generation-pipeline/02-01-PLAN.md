```python
from typing import Annotated, Literal, Union

from pydantic import BaseModel, Field, field_validator

MULTIPLE_CHOICE_OPTIONS_COUNT = 4
TRUE_FALSE_OPTIONS_COUNT = 2
CORRECT_INDICES_COUNT = 1


class BaseQuestion(BaseModel):
    question_type: str
    question: str
    explanation: str


class OptionsQuestion(BaseQuestion):
    options: list[str]
    correct_indices: list[int]


class MultipleChoiceQuestion(OptionsQuestion):
    question_type: Literal["multiple_choice"]

    @field_validator("options")
    @classmethod
    def exactly_four_options(cls, v: list[str]) -> list[str]:
        if len(v) != MULTIPLE_CHOICE_OPTIONS_COUNT:
            raise ValueError(f"MultipleChoiceQuestion requires exactly {MULTIPLE_CHOICE_OPTIONS_COUNT} options, got {len(v)}")
        return v

    @field_validator("correct_indices")
    @classmethod
    def single_correct_index(cls, v: list[int], info) -> list[int]:
        if len(v) != CORRECT_INDICES_COUNT:
            raise ValueError(f"MultipleChoiceQuestion correct_indices must have exactly {CORRECT_INDICES_COUNT} element, got {len(v)}")
        return v


class TrueFalseQuestion(OptionsQuestion):
    question_type: Literal["true_false"]

    @field_validator("options")
    @classmethod
    def exactly_two_options(cls, v: list[str]) -> list[str]:
        if len(v) != TRUE_FALSE_OPTIONS_COUNT:
            raise ValueError(f"TrueFalseQuestion requires exactly {TRUE_FALSE_OPTIONS_COUNT} options, got {len(v)}")
        return v

    @field_validator("correct_indices")
    @classmethod
    def single_correct_index(cls, v: list[int]) -> list[int]:
        if len(v) != CORRECT_INDICES_COUNT:
            raise ValueError(f"TrueFalseQuestion correct_indices must have exactly {CORRECT_INDICES_COUNT} element, got {len(v)}")
        return v


class MultiSelectQuestion(OptionsQuestion):
    question_type: Literal["multi_select"]

    @field_validator("options")
    @classmethod
    def at_least_two_options(cls, v: list[str]) -> list[str]:
        if len(v) < TRUE_FALSE_OPTIONS_COUNT:
            raise ValueError(f"MultiSelectQuestion requires at least {TRUE_FALSE_OPTIONS_COUNT} options")
        return v

    @field_validator("correct_indices")
    @classmethod
    def at_least_one_correct(cls, v: list[int]) -> list[int]:
        if len(v) < CORRECT_INDICES_COUNT:
            raise ValueError("MultiSelectQuestion correct_indices must be non-empty")
        return v


class ShortAnswerQuestion(BaseQuestion):
    question_type: Literal["short_answer"]
    expected_answer: str


QuizQuestion = Annotated[
    Union[
        MultipleChoiceQuestion,
        TrueFalseQuestion,
        MultiSelectQuestion,
        ShortAnswerQuestion,
    ],
    Field(discriminator="question_type"),
]


class QuizSchema(BaseModel):
    questions: list[QuizQuestion]


class QuizResponse(BaseModel):
    questions: list[QuizQuestion]
    model_used: str
    source: Literal["topic", "file"]
    truncated: bool = False
```
